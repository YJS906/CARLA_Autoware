cmake_minimum_required(VERSION 3.16)
project(validate_town05_lane_changes)
set(CMAKE_CXX_STANDARD 17)
find_package(ament_cmake REQUIRED)
find_package(autoware_route_handler REQUIRED)
find_package(autoware_lanelet2_extension REQUIRED)
find_package(lanelet2_io REQUIRED)
find_package(lanelet2_projection REQUIRED)
add_executable(validate_town05_lane_changes validate_town05_lane_changes.cpp)
ament_target_dependencies(validate_town05_lane_changes autoware_route_handler autoware_lanelet2_extension lanelet2_io lanelet2_projection)
add_executable(validate_town05_regulatory validate_town05_regulatory.cpp)
ament_target_dependencies(validate_town05_regulatory autoware_lanelet2_extension lanelet2_io lanelet2_projection)
